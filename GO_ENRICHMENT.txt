# Purpose: Extract genotype-specific genes and create GO mapping
# ============================================================

set -e  # Exit on error

# ============================================================
# CONFIGURATION # ============================================================

# Base directories
ORTHOFINDER_DIR="/home/s2907620/pigeonpea_project/final_proteomes/OrthoFinder/Results_Jul02/Orthogroups"
MASTER_FILES_DIR="/home/s2907620/pigeonpea_project/master_files"
WORK_DIR="$(pwd)"

# Output directory
OUTPUT_DIR="${WORK_DIR}/enrichment_input"

# ============================================================
# SETUP
# ============================================================

echo "=========================================="
echo "PIGEONPEA GO ENRICHMENT PIPELINE"
echo "=========================================="
echo ""
echo "Working directory: $WORK_DIR"
echo "OrthoFinder directory: $ORTHOFINDER_DIR"
echo "Master files directory: $MASTER_FILES_DIR"
echo ""

# Create output directory
mkdir -p "$OUTPUT_DIR"
cd "$OUTPUT_DIR"

echo "Output directory: $OUTPUT_DIR"
echo ""

# ============================================================
# STEP 1: FIND GENOTYPE-SPECIFIC ORTHOGROUPS
# ============================================================

echo "=========================================="
echo "STEP 1: Finding genotype-specific orthogroups"
echo "=========================================="

cat > find_specific_OGs.py << 'EOF'
import pandas as pd

# Load gene count table
df = pd.read_csv("Orthogroups.GeneCount.tsv", sep="\t")

high = ["NSPP_3C_primary", "NSPP_87_primary"]   # HY3C, ICPL87
mod  = ["NSPP_50_primary", "NSPP_70_primary"]   # MC50, MC70

# Find genotype-specific orthogroups
high_specific = df[(df[high] > 0).all(axis=1) & (df[mod] == 0).all(axis=1)]
mod_specific  = df[(df[mod]  > 0).all(axis=1) & (df[high] == 0).all(axis=1)]

print(f"High-protein specific orthogroups: {len(high_specific)}")
print(f"Moderate-protein specific orthogroups: {len(mod_specific)}")

# Save results
high_specific.to_csv("OGs_high_protein_specific.tsv", sep="\t", index=False)
mod_specific.to_csv("OGs_moderate_protein_specific.tsv", sep="\t", index=False)

# Save just the orthogroup IDs
high_specific["Orthogroup"].to_csv("OG_IDs_high_specific.txt", index=False, header=False)
mod_specific["Orthogroup"].to_csv("OG_IDs_moderate_specific.txt", index=False, header=False)

print("✓ Orthogroup files created")
EOF

# Copy OrthoFinder results
cp "$ORTHOFINDER_DIR/Orthogroups.GeneCount.tsv" .
cp "$ORTHOFINDER_DIR/Orthogroups.tsv" .

# Run Python script
python find_specific_OGs.py
echo "✓ Step 1 complete"
echo ""

# ============================================================
# STEP 2: EXTRACT GENES FROM ORTHOGROUPS
# ============================================================

echo "=========================================="
echo "STEP 2: Extracting gene IDs from orthogroups"
echo "=========================================="

awk 'NR==FNR{ogs[$1]; next} FNR==1 || ($1 in ogs)' \
  OG_IDs_high_specific.txt Orthogroups.tsv > genes_high_specific.tsv

awk 'NR==FNR{ogs[$1]; next} FNR==1 || ($1 in ogs)' \
  OG_IDs_moderate_specific.txt Orthogroups.tsv > genes_moderate_specific.tsv

echo "✓ genes_high_specific.tsv created: $(wc -l < genes_high_specific.tsv) lines"
echo "✓ genes_moderate_specific.tsv created: $(wc -l < genes_moderate_specific.tsv) lines"
echo ""

# ============================================================
# STEP 3: EXTRACT INDIVIDUAL GENE IDs BY GENOTYPE
# ============================================================

echo "=========================================="
echo "STEP 3: Extracting individual gene IDs"
echo "=========================================="

# High protein genotypes
awk -F'\t' 'NR>1{print $2}' genes_high_specific.tsv | tr ',' '\n' | sed 's/^ *//;s/ *$//' | grep -v '^$' > geneIDs_HY3C_specific.txt
awk -F'\t' 'NR>1{print $5}' genes_high_specific.tsv | tr ',' '\n' | sed 's/^ *//;s/ *$//' | grep -v '^$' > geneIDs_ICPL87_specific.txt

# Moderate protein genotypes
awk -F'\t' 'NR>1{print $3}' genes_moderate_specific.tsv | tr ',' '\n' | sed 's/^ *//;s/ *$//' | grep -v '^$' > geneIDs_MC50_specific.txt
awk -F'\t' 'NR>1{print $4}' genes_moderate_specific.tsv | tr ',' '\n' | sed 's/^ *//;s/ *$//' | grep -v '^$' > geneIDs_MC70_specific.txt

# Combined gene lists
cat geneIDs_HY3C_specific.txt geneIDs_ICPL87_specific.txt > geneIDs_high_specific.txt
cat geneIDs_MC50_specific.txt geneIDs_MC70_specific.txt > geneIDs_moderate_specific.txt

echo "HY3C genes: $(wc -l < geneIDs_HY3C_specific.txt)"
echo "ICPL87 genes: $(wc -l < geneIDs_ICPL87_specific.txt)"
echo "HP total entries: $(wc -l < geneIDs_high_specific.txt)"
echo ""
echo "MC50 genes: $(wc -l < geneIDs_MC50_specific.txt)"
echo "MC70 genes: $(wc -l < geneIDs_MC70_specific.txt)"
echo "MP total entries: $(wc -l < geneIDs_moderate_specific.txt)"
echo ""

# ============================================================
# STEP 4: GET ANNOTATIONS FROM MASTER FILES
# ============================================================

echo "=========================================="
echo "STEP 4: Extracting annotations from master files"
echo "=========================================="

awk -F'\t' 'NR==FNR{ids[$1]; next} FNR==1 || ($1 in ids)' \
  geneIDs_HY3C_specific.txt \
  "$MASTER_FILES_DIR/NSPP_3C/NSPP_3C_master_annotation.tsv" \
  > HY3C_specific_annotation.tsv

awk -F'\t' 'NR==FNR{ids[$1]; next} FNR==1 || ($1 in ids)' \
  geneIDs_ICPL87_specific.txt \
  "$MASTER_FILES_DIR/NSPP_87/NSPP_87_master_annotation.tsv" \
  > ICPL87_specific_annotation.tsv

awk -F'\t' 'NR==FNR{ids[$1]; next} FNR==1 || ($1 in ids)' \
  geneIDs_MC50_specific.txt \
  "$MASTER_FILES_DIR/NSPP_50/NSPP_50_master_annotated_only.tsv" \
  > MC50_specific_annotation.tsv

awk -F'\t' 'NR==FNR{ids[$1]; next} FNR==1 || ($1 in ids)' \
  geneIDs_MC70_specific.txt \
  "$MASTER_FILES_DIR/NSPP_70/NSPP_70_master_annotated.tsv" \
  > MC70_specific_annotation.tsv

echo "HY3C annotated: $(wc -l < HY3C_specific_annotation.tsv) lines"
echo "ICPL87 annotated: $(wc -l < ICPL87_specific_annotation.tsv) lines"
echo "MC50 annotated: $(wc -l < MC50_specific_annotation.tsv) lines"
echo "MC70 annotated: $(wc -l < MC70_specific_annotation.tsv) lines"
echo "✓ Step 4 complete"
echo ""

# ============================================================
# STEP 5: COMBINE ANNOTATIONS BY PHENOTYPE
# ============================================================

echo "=========================================="
echo "STEP 5: Combining annotations by phenotype"
echo "=========================================="

# Combine HP genotypes
(head -1 HY3C_specific_annotation.tsv && \
 tail -n +2 HY3C_specific_annotation.tsv && \
 tail -n +2 ICPL87_specific_annotation.tsv) \
> HighProtein_annotation.tsv

# Combine MP genotypes
(head -1 MC50_specific_annotation.tsv && \
 tail -n +2 MC50_specific_annotation.tsv && \
 tail -n +2 MC70_specific_annotation.tsv) \
> ModerateProtein_annotation.tsv

echo "HighProtein_annotation.tsv: $(wc -l < HighProtein_annotation.tsv) lines"
echo "ModerateProtein_annotation.tsv: $(wc -l < ModerateProtein_annotation.tsv) lines"
echo "✓ Step 5 complete"
echo ""

# ============================================================
# STEP 6: FILTER TRANSPOSABLE ELEMENTS
# ============================================================

echo "=========================================="
echo "STEP 6: Filtering transposable elements"
echo "=========================================="

grep -Eiv "transposon|retrotransposon|reverse transcriptase|retrovirus|gag-pol|integrase|transposable" \
  HighProtein_annotation.tsv > HighProtein_noTE.tsv

grep -Eiv "transposon|retrotransposon|reverse transcriptase|retrovirus|gag-pol|integrase|transposable" \
  ModerateProtein_annotation.tsv > ModerateProtein_noTE.tsv

echo "HighProtein after TE filter: $(wc -l < HighProtein_noTE.tsv) lines"
echo "ModerateProtein after TE filter: $(wc -l < ModerateProtein_noTE.tsv) lines"
echo "✓ Step 6 complete"
echo ""

# ============================================================
# STEP 7: EXTRACT GO TERMS
# ============================================================

echo "=========================================="
echo "STEP 7: Extracting GO terms"
echo "=========================================="

# Determine which column has GO terms (usually last column)
GO_COL=$(head -1 HighProtein_noTE.tsv | tr '\t' '\n' | wc -l)

echo "Total columns in annotation file: $GO_COL"
echo "Using column $GO_COL for GO terms"
echo ""

# Extract GO terms (last column = NF)
awk -F'\t' 'NR>1 && NF>0 && $NF != "-" && $NF != "" && $NF != "NA" {
  split($NF, gos, ",")
  for(i in gos) {
    gsub(/^ +| +$/, "", gos[i])
    if(gos[i] != "") print $1 "\t" gos[i]
  }
}' HighProtein_noTE.tsv > HighProtein_gene_GO.tsv

awk -F'\t' 'NR>1 && NF>0 && $NF != "-" && $NF != "" && $NF != "NA" {
  split($NF, gos, ",")
  for(i in gos) {
    gsub(/^ +| +$/, "", gos[i])
    if(gos[i] != "") print $1 "\t" gos[i]
  }
}' ModerateProtein_noTE.tsv > ModerateProtein_gene_GO.tsv

echo "HighProtein_gene_GO.tsv: $(wc -l < HighProtein_gene_GO.tsv) lines"
echo "ModerateProtein_gene_GO.tsv: $(wc -l < ModerateProtein_gene_GO.tsv) lines"
echo "✓ Step 7 complete"
echo ""

# ============================================================
# STEP 8: EXTRACT UNIQUE GENES
# ============================================================

echo "=========================================="
echo "STEP 8: Extracting unique genes"
echo "=========================================="

# Get unique genes (this is CRITICAL for R script)
cut -f1 HighProtein_gene_GO.tsv | sort -u > hp_genes.txt
cut -f1 ModerateProtein_gene_GO.tsv | sort -u > mp_genes.txt

HP_COUNT=$(wc -l < hp_genes.txt)
MP_COUNT=$(wc -l < mp_genes.txt)

echo "Unique HP genes: $HP_COUNT"
echo "Unique MP genes: $MP_COUNT"

if [ "$HP_COUNT" -ne 264 ]; then
  echo "WARNING: HP gene count is $HP_COUNT (expected 264)"
fi

if [ "$MP_COUNT" -ne 370 ]; then
  echo "WARNING: MP gene count is $MP_COUNT (expected 370)"
fi

echo "✓ Step 8 complete"
echo ""

# ============================================================
# STEP 9: CREATE GO-GENE BACKGROUND MAPPING
# ============================================================

echo "=========================================="
echo "STEP 9: Creating GO-gene background mapping"
echo "=========================================="

echo "Merging all master annotations..."

{
head -1 "$MASTER_FILES_DIR/NSPP_3C/NSPP_3C_master_annotation.tsv"
tail -n +2 "$MASTER_FILES_DIR/NSPP_3C/NSPP_3C_master_annotation.tsv"
tail -n +2 "$MASTER_FILES_DIR/NSPP_50/NSPP_50_master_annotation.tsv"
tail -n +2 "$MASTER_FILES_DIR/NSPP_70/NSPP_70_master_annotation.tsv"
tail -n +2 "$MASTER_FILES_DIR/NSPP_87/NSPP_87_master_annotation.tsv"
} > merged_master_annotation.tsv

echo "merged_master_annotation.tsv: $(wc -l < merged_master_annotation.tsv) lines"
echo ""

echo "Creating go2gene.tsv (GO_term TAB gene_id format)..."

# CRITICAL: GO term FIRST, gene_id SECOND
# Extract from second-to-last column (GO terms) and second column (gene_id)
awk -F'\t' 'NR>1 && $(NF-1) != "-" && $(NF-1) != "" && $(NF-1) != "NA" {
  gsub(/^ +| +$/, "", $(NF-1))
  split($(NF-1), gos, ",")
  for(i in gos) {
    gsub(/^ +| +$/, "", gos[i])
    if(gos[i] != "") print gos[i] "\t" $2
  }
}' merged_master_annotation.tsv > go2gene.tsv

GO_LINES=$(wc -l < go2gene.tsv)
echo "go2gene.tsv: $GO_LINES lines"
echo "✓ Step 9 complete"
echo ""

# ============================================================
# STEP 10: VERIFICATION
# ============================================================

echo "=========================================="
echo "STEP 10: Verification"
echo "=========================================="

echo ""
echo "Checking go2gene.tsv format (should be GO_term TAB gene_id):"
echo "Column 1 (GO terms - first 5):"
cut -f1 go2gene.tsv | head -5
echo ""
echo "Column 2 (gene IDs - first 5):"
cut -f2 go2gene.tsv | head -5
echo ""

echo "Sample hp_genes.txt (first 10):"
head -10 hp_genes.txt
echo ""

echo "Sample mp_genes.txt (first 10):"
head -10 mp_genes.txt
echo ""

# ============================================================
# STEP 11: FINAL SUMMARY
# ============================================================

echo "=========================================="

echo "=========================================="

echo "Generated files:"
echo "  1. hp_genes.txt ($HP_COUNT genes) ✓"

echo ""

cat("\n--- Adding GO function names ---\n")

get_go_function <- function(go_id) {
  tryCatch({
    term_obj <- GOTERM[[go_id]]
    if(!is.null(term_obj)) {
      return(as.character(Term(term_obj)))
    } else {
      return(go_id)
    }
  }, error = function(e) {
    return(go_id)
  })
}

hp_res$Function <- sapply(hp_res$ID, get_go_function)
mp_res$Function <- sapply(mp_res$ID, get_go_function)
cat("✓ Function names added\n\n")

# ============================================================
# STEP 6: ADD GO ONTOLOGY (BEFORE FILTERING)
# ============================================================

cat("\n--- Extracting GO ontology categories ---\n")

get_go_ontology <- function(go_id) {
  tryCatch({
    term_obj <- GOTERM[[go_id]]
    if(!is.null(term_obj)) {
      return(as.character(Ontology(term_obj)))
    } else {
      return("Unknown")
    }
  }, error = function(e) {
    return("Unknown")
  })
}

hp_res$Ontology <- sapply(hp_res$ID, get_go_ontology)
mp_res$Ontology <- sapply(mp_res$ID, get_go_ontology)

wrap_text <- function(text, width=45) {
  paste(strwrap(text, width=width), collapse="\n")
}

prepare_plot_data <- function(res_df, top_n=20) {
  res_df %>%
    arrange(p.adjust) %>%
    head(top_n) %>%
    mutate(
      Function_with_count = paste0(Function, " (", Count, ")"),
      Function_wrapped = sapply(Function_with_count, wrap_text, width=45)
    ) %>%
    arrange(desc(p.adjust))
}

    )

create_gb_plot <- function(plot_data, title_text, x_max, x_breaks) {
  ggplot(plot_data, aes(x=RichFactor, y=reorder(Function_wrapped, RichFactor))) +
    geom_point(aes(size=Count, color=p.adjust), alpha=0.75, stroke=0.6) +
    scale_color_gradient(
      low="#0072B2",
      high="#E31A1C",
      name="q-value",
      trans="log10",
      limits=c(1e-15, 1e-06),
      breaks=c(1e-15, 1e-12, 1e-09, 1e-06),
      labels=c("1e-15", "1e-12", "1e-09", "1e-06"),
      oob=squish
    ) +
    scale_size_continuous(
      name="Gene Count",
      range=c(2, 8),
      breaks=c(5, 10, 15, 20, 25)
    ) +
    scale_x_continuous(
      limits=c(0, x_max),
      breaks=x_breaks
    ) +
    theme_genome_biology() +
    guides(
      size = guide_legend(order=1, title="Gene Count"),
      color = guide_colorbar(order=2, title="q-value")
    ) +
    labs(title=title_text, x="Rich Factor", y="")
}

cat("\n--- Creating gene-centric TSV files ---\n")

expand_genes <- function(res_df, genotype_name) {
  gene_functions <- data.frame()
  
  for(i in 1:nrow(res_df)) {
    go_id <- res_df$ID[i]
    function_name <- res_df$Function[i]
    ontology <- res_df$Ontology[i]
    genes <- strsplit(res_df$geneID[i], "/")[[1]]
    
    for(gene in genes) {
      gene_functions <- rbind(gene_functions, data.frame(
        Gene_ID = gene,
        GO_ID = go_id,
        GO_Function = function_name,
        Ontology = ontology,
        Genotype = genotype_name,
        Count_in_GO = res_df$Count[i],
        RichFactor = res_df$RichFactor[i],
        FoldEnrichment = res_df$FoldEnrichment[i],
        p.adjust = res_df$p.adjust[i]
      ))
    }
  }
  
  return(gene_functions)
}
