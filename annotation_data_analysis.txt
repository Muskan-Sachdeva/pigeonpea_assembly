species_legume <- master_87 %>%
  filter(!is.na(species), species != "") %>%
  mutate(
    species_lower = tolower(species),
    is_legume = str_detect(
      species_lower,
      paste(legume_keywords, collapse = "|")
    )
  )


####count unique genes
