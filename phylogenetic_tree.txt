##Step3: Alignment trimming using TrimAl

#!/bin/bash
# ============================================================
# trimAl trimming script for 270 single-copy orthogroup alignments
# Purpose: Remove poorly aligned/noisy columns before concatenation
#
# Parameters used:
#   -in            : input MAFFT-aligned FASTA file
#   -out           : output trimmed alignment file
#   -automated1    : automated heuristic mode — trimAl decides
#                    best trimming strategy based on gap
#                    distribution + conservation pattern.
#                    Standard/default choice in phylogenomics
#                    papers, no manual threshold tuning needed.
# ============================================================

INPUT_DIR="/home/s2907620/pigeonpea_project/phylogenetic_tree/mafft_alignment"
OUTPUT_DIR="/home/s2907620/pigeonpea_project/phylogenetic_tree/trimmed_orthogroups"
mkdir -p "$OUTPUT_DIR"

cd "$INPUT_DIR"

for f in *_aligned.fa; do
    base=$(basename "$f" _aligned.fa)
    trimal -in "$f" -out "$OUTPUT_DIR/${base}_trimmed.fa" -automated1
done

echo "Trimming complete. Trimmed files saved in: $OUTPUT_DIR"
ls "$OUTPUT_DIR" | wc -l


################################################################################
