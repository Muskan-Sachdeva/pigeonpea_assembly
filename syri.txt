# Load environment modules and activate environment
. /etc/profile.d/modules.sh
conda activate Syri

# ========= Input files =========
REF="reference_genome.fasta"
QRY="query_assembly_trimmed.fasta"
ALIGN="syri_cigar.paf"  # PAF file generated by minimap2 -c --eqx

# ========= Output settings =========
WORKDIR="syri_results_query_name"
mkdir -p "$WORKDIR"
PREFIX="query_name"

# ========= Run SyRI =========
# -c: alignment file (PAF)
# -F P: input is PAF format
# -r: reference genome
# -q: query genome
# --dir: output directory
# --prefix: output file prefix (e.g., prefix.syri.out)
syri -c "$ALIGN" \
     -F P \
     -r "$REF" \
     -q "$QRY" \
     --dir "$WORKDIR" \
     --prefix "$PREFIX" \
     -k \
     --log INFO

############## RUN PLOTSR############
plotsr \
  --sr syri_results_A/A_syri.out \
  --sr syri_results_B/B_syri.out \
  --sr syri_results_C/C_syri.out \
  --genomes genomes.txt \
  -o all_synteny_plot.png
